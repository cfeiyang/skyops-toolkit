```python
def current_local_day_period() -> tuple[str, str]:
    """按电脑本地日期生成当天 00:00 至次日 00:00 的 UTCG 文本。"""
    start = datetime.now().replace(hour=0, minute=0, second=0, microsecond=0)
    stop = start + timedelta(days=1)
    return _stk_time(start), _stk_time(stop)


def _stk_time(value: datetime) -> str:
    months = (
        "Jan", "Feb", "Mar", "Apr", "May", "Jun",
        "Jul", "Aug", "Sep", "Oct", "Nov", "Dec",
    )
    return (
        f"{value.day} {months[value.month - 1]} {value.year} "
        f"{value:%H:%M:%S}.000"
    )


def parse_stk_time(text: str) -> datetime:
    """将 STK UTCG 时间字符串 (如 '1 Sep 2026 00:00:00.000') 或 ISO 格式解析为 UTC datetime。"""
    cleaned = text.strip()
    for fmt in ("%d %b %Y %H:%M:%S.%f", "%d %b %Y %H:%M:%S", "%Y-%m-%d %H:%M:%S"):
        try:
            return datetime.strptime(cleaned, fmt).replace(tzinfo=timezone.utc)
        except ValueError:
            continue
    raise ValueError(f"无法识别的时间格式: {text}")
```

#### 1.3 修改 `src/stk_simulation/stk/session.py`
将 `from agi.stk12.stkdesktop import STKDesktop` 移入 `connect()` 方法内部，确保在未安装 STK 的电脑上使用纯 Python 引擎时绝对不会报 `ModuleNotFoundError`：

```python
"""STK Desktop 会话和 Scenario 生命周期（支持无 STK 环境安全导入）。"""

from __future__ import annotations


class STKSession:
    """在创建它的工作线程中持有并复用 STK Desktop 对象。"""

    def __init__(self) -> None:
        self.application = None
        self.root = None
        self.mode: str | None = None

    def connect(self, mode: str = "new"):
        try:
            from agi.stk12.stkdesktop import STKDesktop  # type: ignore
        except ImportError as exc:
            raise RuntimeError(
                "当前环境未安装 STK 12 Python API (agi.stk12)。"
                "请切换至顶部的「⚡ 纯 Python 极速引擎」运行，或在装有 STK 12.6 的环境中运行。"
            ) from exc

        normalized_mode = str(mode).lower()
        if self.root is not None and normalized_mode == self.mode:
            return self.root

        if normalized_mode == "new":
            print("[STK] 正在新建 STK 12 实例...")
            application = STKDesktop.StartApplication(visible=True, userControl=True)
        elif normalized_mode == "running":
            print("[STK] 正在连接已运行的 STK 12 实例...")
            application = STKDesktop.AttachToApplication()
        else:
            raise ValueError(
                'STKSession.connect: mode 必须是 "new" 或 "running"，'
                f"收到：{mode}"
            )

        self.application = application
        self.root = application.Root
        self.mode = normalized_mode
        print("[STK] 对象根 root 获取成功。")
        return self.root

    def release(self) -> None:
        """释放 Python 侧引用；由用户控制的 STK Desktop 保持打开。"""
        self.root = None
        self.application = None
        self.mode = None

    @staticmethod
    def create_scenario(root, name: str, start_time: str, stop_time: str):
        preferences = root.UnitPreferences
        preferences.Item("AngleUnit").SetCurrentUnit("deg")
        preferences.Item("DistanceUnit").SetCurrentUnit("km")
        preferences.Item("TimeUnit").SetCurrentUnit("sec")
        preferences.Item("DateFormat").SetCurrentUnit("UTCG")
        preferences.Item("FrequencyUnit").SetCurrentUnit("MHz")

        if root.CurrentScenario is not None:
            root.CloseScenario()

        root.NewScenario(name)
        scenario = root.CurrentScenario
        scenario.SetTimePeriod(start_time, stop_time)
        root.Rewind()
        print(f"[场景] 成功创建 Scenario: {name}")
        return scenario
```

#### 2.1 新增 `src/stk_simulation/simulations/global_coverage/py_engine.py`

```python
"""纯 Python 向量化全球通信覆盖计算引擎（免 STK 商业授权，支持多星几何中继）。"""

from __future__ import annotations

from datetime import datetime, timedelta, timezone
from math import asin, atan2, cos, degrees, pi, radians, sin, sqrt
from typing import Callable

import numpy as np
import pandas as pd
from sgp4.api import Satrec, WGS72, jday

from stk_simulation.infrastructure.tle import TLEData, load_latest_tle
from stk_simulation.simulations.common import OutputResult, parse_stk_time
from stk_simulation.simulations.global_coverage.config import GlobalCoverageConfig
from stk_simulation.simulations.global_coverage.presets import derived_satellite_elements
from stk_simulation.simulations.global_coverage.service import (
    GlobalCoverageResult,
    GlobalCoverageSummary,
)

EARTH_RADIUS_KM = 6378.137
EARTH_FLAT = 1.0 / 298.257223563
ISL_MIN_TANGENT_ALT_KM = 80.0  # 星间链路最低切线高度（避开稠密大气与地表遮挡）


def _gmst_rad(jd_ut1: float) -> np.ndarray | float:
    """IAU 1982 格林尼治恒星时 (弧度)。"""
    t = (jd_ut1 - 2451545.0) / 36525.0
    sec = (
        67310.54841
        + (876600.0 * 3600.0 + 8640184.812866) * t
        + 0.093104 * (t ** 2)
        - 6.2e-6 * (t ** 3)
    )
    return np.deg2rad((sec / 240.0) % 360.0)


def _teme_to_ecef(r_teme: np.ndarray, jd_array: np.ndarray) -> np.ndarray:
    """将 (T, 3) 的 TEME 坐标旋转为 ECEF 地固坐标 (km)。"""
    theta = _gmst_rad(jd_array)
    c = np.cos(theta)
    s = np.sin(theta)
    x = c * r_teme[:, 0] + s * r_teme[:, 1]
    y = -s * r_teme[:, 0] + c * r_teme[:, 1]
    z = r_teme[:, 2]
    return np.stack([x, y, z], axis=-1)


def _geodetic_to_ecef_enu(
    lat_deg: np.ndarray, lon_deg: np.ndarray, alt_m: np.ndarray
) -> tuple[np.ndarray, np.ndarray]:
    """将地理经纬高转换为 ECEF 坐标 (N, 3) 与向上法向量 Up (N, 3)。"""
    lat = np.deg2rad(lat_deg)
    lon = np.deg2rad(lon_deg)
    alt_km = alt_m / 1000.0
    e2 = EARTH_FLAT * (2.0 - EARTH_FLAT)
    sin_lat = np.sin(lat)
    cos_lat = np.cos(lat)
    cos_lon = np.cos(lon)
    sin_lon = np.sin(lon)

    n_rad = EARTH_RADIUS_KM / np.sqrt(1.0 - e2 * sin_lat * sin_lat)
    x = (n_rad + alt_km) * cos_lat * cos_lon
    y = (n_rad + alt_km) * cos_lat * sin_lon
    z = (n_rad * (1.0 - e2) + alt_km) * sin_lat
    pos = np.stack([x, y, z], axis=-1)

    up = np.stack([cos_lat * cos_lon, cos_lat * sin_lon, sin_lat], axis=-1)
    return pos, up


def _ecef_to_latlon(r_ecef: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    """ECEF (T, 3) 转星下点纬度、经度 (度)。"""
    x, y, z = r_ecef[:, 0], r_ecef[:, 1], r_ecef[:, 2]
    lon = np.rad2deg(np.arctan2(y, x))
    r_xy = np.hypot(x, y)
    lat = np.rad2deg(np.arctan2(z, r_xy * (1.0 - EARTH_FLAT) ** 2))
    return lat, lon


def _build_satrec_from_tle_file(tle_path) -> Satrec:
    lines = [
        line.strip()
        for line in tle_path.read_text(encoding="ascii").splitlines()
        if line.strip()
    ]
    l1 = next(line for line in lines if line.startswith("1 "))
    l2 = next(line for line in lines if line.startswith("2 "))
    return Satrec.twoline2rv(l1, l2, WGS72)


def _build_satrec_from_elements(elem, ref_sat: Satrec) -> Satrec:
    """利用参考星的阻尼项与轨道根数构造派生卫星的 SGP4 算子。"""
    sat = Satrec()
    sat.sgp4init(
        WGS72,
        "i",
        elem.ssc,
        ref_sat.jdsatepoch - 2433281.5,  # sgp4 2.x 的 jdsatepoch 已含小数日，勿再加 jdsatepochF
        ref_sat.bstar,
        ref_sat.ndot,
        ref_sat.nddot,
        elem.eccentricity,
        radians(elem.arg_perigee),
        radians(elem.inclination),
        radians(elem.mean_anomaly),
        radians(elem.mean_motion) * 60.0,  # 度/秒 -> 弧度/分（presets 的 mean_motion 单位是 度/秒）
        radians(elem.raan),
    )
    return sat


def _isl_visible_matrix(sat_ecef: np.ndarray) -> np.ndarray:
    """计算 (T, S, S) 布尔矩阵：判断任意两颗卫星之间是否存在无地球遮挡的几何视线。"""
    t_steps, s_count, _ = sat_ecef.shape
    isl = np.zeros((t_steps, s_count, s_count), dtype=bool)
    min_r2 = (EARTH_RADIUS_KM + ISL_MIN_TANGENT_ALT_KM) ** 2

    for i in range(s_count):
        for j in range(i + 1, s_count):
            r1 = sat_ecef[:, i, :]  # (T, 3)
            r2 = sat_ecef[:, j, :]  # (T, 3)
            d = r2 - r1
            d_norm2 = np.einsum("ij,ij->i", d, d)
            # 线段 R1 + u*(R2 - R1) 离地心最近点的参数 u
            u = -np.einsum("ij,ij->i", r1, d) / np.maximum(d_norm2, 1e-9)
            u_clamped = np.clip(u, 0.0, 1.0)[:, None]
            closest = r1 + u_clamped * d
            closest_r2 = np.einsum("ij,ij->i", closest, closest)
            ok = closest_r2 >= min_r2
            isl[:, i, j] = ok
            isl[:, j, i] = ok
    return isl


def compute_global_coverage_python(
    config: GlobalCoverageConfig,
    *,
    tle: TLEData | None = None,
    progress: Callable[[str], None] | None = None,
) -> GlobalCoverageResult:
    """纯 Python 向量化计算全球通信覆盖时长，输出与 STK 完全同构的 GlobalCoverageResult。"""
    if progress:
        progress("[Python 引擎] 正在加载 TLE 轨道根数...")
    tle = tle or load_latest_tle(
        config.satellite.tle_url,
        config.paths.data_dir,
        config.satellite.ssc,
    )

    start_dt = parse_stk_time(config.scenario.start_time)
    stop_dt = parse_stk_time(config.scenario.stop_time)
    step_sec = float(max(config.satellite.step, 10.0))
    total_sec = max((stop_dt - start_dt).total_seconds(), step_sec)
    num_steps = int(total_sec // step_sec) + 1

    times = [start_dt + timedelta(seconds=i * step_sec) for i in range(num_steps)]
    jd_list, fr_list = zip(
        *(
            jday(dt.year, dt.month, dt.day, dt.hour, dt.minute, dt.second + dt.microsecond / 1e6)
            for dt in times
        )
    )
    jd_arr = np.asarray(jd_list, dtype=float)
    fr_arr = np.asarray(fr_list, dtype=float)
    jd_full = jd_arr + fr_arr

    # 1. 构建卫星列表并传播轨迹
    ref_sat = _build_satrec_from_tle_file(tle.path)
    sat_models = [(config.satellite.name, ref_sat)]
    if config.satellite_group.mode != "single":
        for elem in derived_satellite_elements(tle, step_sec):
            sat_models.append((elem.name, _build_satrec_from_elements(elem, ref_sat)))

    s_count = len(sat_models)
    sat_ecef = np.zeros((num_steps, s_count, 3), dtype=float)
    tracks: list[tuple[np.ndarray, np.ndarray]] = []

    if progress:
        progress(f"[Python 引擎] 正在外推 {s_count} 颗卫星的 SGP4 轨道位置...")

    for s_idx, (_, sat_obj) in enumerate(sat_models):
        err, r_teme, _ = sat_obj.sgp4_array(jd_arr, fr_arr)
        if np.any(err != 0):
            raise RuntimeError(f"SGP4 轨道外推异常，错误码: {np.unique(err)}")
        ecef = _teme_to_ecef(r_teme, jd_full)
        sat_ecef[:, s_idx, :] = ecef

        lat, lon = _ecef_to_latlon(ecef)
        lon_wrapped = np.mod(lon - config.plot.lon_range[0], 360.0) + config.plot.lon_range[0]
        jumps = np.concatenate(([False], np.abs(np.diff(lon_wrapped)) > 180.0))
        tracks.append((np.where(jumps, np.nan, lat), np.where(jumps, np.nan, lon_wrapped)))

    # 2. 计算地面站对各卫星的可见性 (T, S)
    st_lats = np.array([st.lat for st in config.stations], dtype=float)
    st_lons = np.array([st.lon for st in config.stations], dtype=float)
    st_alts = np.array([st.alt for st in config.stations], dtype=float)
    st_min_sin = np.sin(np.deg2rad([st.min_elevation for st in config.stations]))
    st_pos, st_up = _geodetic_to_ecef_enu(st_lats, st_lons, st_alts)  # (K, 3)

    # 相对矢量: (T, S, K, 3)
    diff_st = sat_ecef[:, :, None, :] - st_pos[None, None, :, :]
    dist_st = np.linalg.norm(diff_st, axis=-1)
    sin_el_st = np.einsum("tski,ki->tsk", diff_st, st_up) / np.maximum(dist_st, 1e-9)
    st_sat_vis = sin_el_st >= st_min_sin[None, None, :]  # (T, S, K)
    any_station_vis = np.any(st_sat_vis, axis=-1)  # (T, S) 任意地面站可见该卫星

    # 3. 若开启多星几何星间中继，计算每颗卫星是否能连通地面站 (直达或经过 1 跳星间链路)
    if config.satellite_group.mode == "single" or s_count == 1:
        sat_connected_to_ground = any_station_vis[:, :1]  # (T, 1)
    else:
        isl_ok = _isl_visible_matrix(sat_ecef)  # (T, S, S)
        relay_vis = np.any(isl_ok & any_station_vis[:, None, :], axis=-1)  # (T, S)
        sat_connected_to_ground = any_station_vis | relay_vis  # (T, S)

    # 4. 生成全球覆盖网格并分块向量化计算覆盖时长
    res = float(max(config.coverage.resolution, 0.25))
    lats = np.arange(-90.0, 90.0 + 1e-6, res)
    lons = np.arange(-180.0, 180.0, res)
    lon_grid, lat_grid = np.meshgrid(lons, lats)
    flat_lats = lat_grid.ravel()
    flat_lons = lon_grid.ravel()
    total_points = flat_lats.size

    grid_pos, grid_up = _geodetic_to_ecef_enu(
        flat_lats, flat_lons, np.zeros_like(flat_lats)
    )
    min_sin_grid = sin(radians(config.coverage.min_elevation))
    min_assets = max(int(config.fom.min_assets), 1)

    covered_steps = np.zeros(total_points, dtype=np.int32)
    chunk_size = 1800  # 分块广播，兼顾极速计算与低内存占用
    active_sats = 1 if config.satellite_group.mode == "single" else s_count

    for start_idx in range(0, total_points, chunk_size):
        end_idx = min(start_idx + chunk_size, total_points)
        if progress:
            pct = int(end_idx * 100 / total_points)
            progress(f"[Python 引擎] 正在解算全球网格覆盖 ({pct}%)...")

        g_pos = grid_pos[start_idx:end_idx]  # (C, 3)
        g_up = grid_up[start_idx:end_idx]    # (C, 3)

        # (T, S, C, 3)
        d_vec = sat_ecef[:, :active_sats, None, :] - g_pos[None, None, :, :]
        d_norm = np.linalg.norm(d_vec, axis=-1)
        sin_el = np.einsum("tsci,ci->tsc", d_vec, g_up) / np.maximum(d_norm, 1e-9)
        grid_sees_sat = sin_el >= min_sin_grid  # (T, S, C)

        # 端到端链路导通 = 网格点看见卫星 & 该卫星此刻连通地面站
        end_to_end = grid_sees_sat & sat_connected_to_ground[:, :active_sats, None]
        active_count = np.sum(end_to_end, axis=1)  # (T, C)
        point_active = active_count >= min_assets   # (T, C)
        covered_steps[start_idx:end_idx] = np.sum(point_active, axis=0)

    # 修正首尾时间步权重
    time_sec = covered_steps.astype(float) * (total_sec / max(num_steps - 1, 1))
    time_sec = np.minimum(time_sec, total_sec)
    time_hour = time_sec / 3600.0

    coverage_df = pd.DataFrame(
        {
            "LatitudeDeg": flat_lats,
            "LongitudeDeg": np.mod(flat_lons, 360.0),
            "TimeSec": time_sec,
            "TimeHour": time_hour,
        }
    )

    sat_names = [name for name, _ in sat_models[:active_sats]]
    st_names = [st.name for st in config.stations]
    image_name = config.output.image_name or (
        f"py_global_coverage_{len(sat_names)}sat_{len(st_names)}station.png"
    )
    config.output.image_name = image_name
    config.plot.title = (
        f"[Python 引擎] {'/'.join(sat_names)}-{'/'.join(st_names)} "
        f"全球通信时长（{len(sat_names)}星×{len(st_names)}站）"
    )
    image_path = config.paths.result_dir / image_name

    return GlobalCoverageResult(
        coverage_data=coverage_df,
        satellite_tracks=tracks[:active_sats],
        summary=GlobalCoverageSummary(
            satellite_count=len(sat_names),
            station_count=len(st_names),
            max_hour=float(coverage_df["TimeHour"].max(skipna=True)),
        ),
        output=OutputResult(config.paths.result_dir, image_name, image_path),
        cfg=config,
    )
```

```python
            (parse_stk_time(str(s)), parse_stk_time(str(e)))

        ]

        access.ComputeAccess()

            el = numeric_dataset(res, "Elevation")
            rg = numeric_dataset(res, "Range")

            return []

                {

                    "azimuth_deg": float(az[idx]),
                    "elevation_deg": float(el[idx]),
```

```python
"""基于 Skyfield SGP4 + 二分法的星地建链可见性纯 Python 计算引擎。"""

from __future__ import annotations

from datetime import datetime, timedelta, timezone
from typing import Sequence
from zoneinfo import ZoneInfo

import numpy as np
from skyfield.api import EarthSatellite, Loader, wgs84

from stk_simulation.infrastructure.tle import load_latest_tle
from stk_simulation.simulations.common import OutputResult, parse_stk_time
from stk_simulation.simulations.link_visibility.config import (
    AERRecord,
    AccessWindow,
    LinkStationConfig,
    LinkVisibilityConfig,
    LinkVisibilityResult,
    TrackPoint,
)

LOCAL_TZ = ZoneInfo("Asia/Shanghai")
TIME_FMT = "%Y-%m-%d %H:%M:%S"
GEO_RADIUS_KM = 42164.0
EARTH_RADIUS_KM = 6371.0


def _fmt_bj(dt_utc: datetime) -> str:
    return dt_utc.astimezone(LOCAL_TZ).strftime(TIME_FMT)


def _gso_arc(station: LinkStationConfig, step_deg: float = 2.0) -> np.ndarray:
    lat = np.radians(station.latitude_deg)
    lon = np.radians(station.longitude_deg)
    sx = EARTH_RADIUS_KM * np.cos(lat) * np.cos(lon)
    sy = EARTH_RADIUS_KM * np.cos(lat) * np.sin(lon)
    sz = EARTH_RADIUS_KM * np.sin(lat)

    east = np.array([-np.sin(lon), np.cos(lon), 0.0])
    north = np.array([-np.sin(lat) * np.cos(lon), -np.sin(lat) * np.sin(lon), np.cos(lat)])
    up = np.array([np.cos(lat) * np.cos(lon), np.cos(lat) * np.sin(lon), np.sin(lat)])

    rows = []
    for g_lon in np.arange(-180.0, 180.0, step_deg):
        gl = np.radians(float(g_lon))
        d = np.array([GEO_RADIUS_KM * np.cos(gl) - sx, GEO_RADIUS_KM * np.sin(gl) - sy, -sz])
        dist = float(np.linalg.norm(d))
        if dist <= 0.0:
            continue
        el = float(np.degrees(np.arcsin(np.clip(float(d @ up) / dist, -1.0, 1.0))))
        if el <= 0.0:
            continue
        az = float(np.degrees(np.arctan2(float(d @ east), float(d @ north))) % 360.0)
        rows.append((az, el))
    return np.array(rows, dtype=float) if rows else np.zeros((0, 2), dtype=float)


def _gso_sep_deg(az_deg, el_deg, arc: np.ndarray) -> np.ndarray:
    az = np.radians(np.asarray(az_deg, dtype=float))
    el = np.radians(np.asarray(el_deg, dtype=float))
    if arc is None or len(arc) == 0:
        return np.full(np.shape(az), 90.0)
    arc_az = np.radians(arc[:, 0])
    arc_el = np.radians(arc[:, 1])
    cos_sep = (
        np.sin(el)[..., None] * np.sin(arc_el)[None, ...]
        + np.cos(el)[..., None] * np.cos(arc_el)[None, ...] * np.cos(az[..., None] - arc_az[None, ...])
    )
    return np.squeeze(np.degrees(np.arccos(np.clip(cos_sep, -1.0, 1.0))).min(axis=-1))


def _mask_elevation(az_deg, base_min: float, mask: Sequence[tuple[float, float, float]]):
    az = np.asarray(az_deg, dtype=float)
    eff = np.full(np.shape(az), float(base_min), dtype=float)
    for s, e, el in mask:
        s, e = float(s) % 360.0, float(e) % 360.0
        if abs(s - e) < 1e-9:
            continue
        inside = ((az >= s) & (az < e)) if s < e else ((az >= s) | (az < e))
        eff = np.maximum(eff, np.where(inside, float(el), -1e9))
    return eff


class _StationHandle:
    def __init__(self, ts, sat, st: LinkStationConfig) -> None:
        self.station = st
        self.ts = ts
        self.topocentric = sat - wgs84.latlon(st.latitude_deg, st.longitude_deg, elevation_m=st.altitude_m)
        self.gso_arc = _gso_arc(st) if st.gso_avoid_deg > 0 else np.zeros((0, 2))

    def altaz(self, t):
        return self.topocentric.at(t).altaz()

    def visible_at(self, t):
        alt, az, _ = self.altaz(t)
        min_el = _mask_elevation(az.degrees, self.station.min_elevation_deg, self.station.mask)
        ok = alt.degrees >= min_el
        if self.station.gso_avoid_deg > 0 and len(self.gso_arc):
            sep = _gso_sep_deg(az.degrees, alt.degrees, self.gso_arc)
            ok = ok & (sep >= self.station.gso_avoid_deg)
        return ok


def compute_link_visibility_python(config: LinkVisibilityConfig) -> LinkVisibilityResult:
    """运行纯 Python 星地建链与 AER 计算。"""
    tle = load_latest_tle(
        config.satellite.tle_url,
        config.paths.data_dir,
        config.satellite.ssc,
    )
    lines = [ln.strip() for ln in tle.path.read_text(encoding="ascii").splitlines() if ln.strip()]
    line1 = next(ln for ln in lines if ln.startswith("1 "))
    line2 = next(ln for ln in lines if ln.startswith("2 "))

    active_stations = config.stations[:2] if config.dual_station_sync else config.stations[:1]
    loader = Loader(str(config.paths.data_dir))
    ts = loader.timescale(builtin=True)
    sat = EarthSatellite(line1, line2, config.satellite.name, ts)
    handles = [_StationHandle(ts, sat, st) for st in active_stations]

    start_utc = parse_stk_time(config.scenario.start_time)
    stop_utc = parse_stk_time(config.scenario.stop_time)
    span_sec = max((stop_utc - start_utc).total_seconds(), 60.0)
    step = float(max(config.coarse_step_sec, 5.0))
    grid = [start_utc + timedelta(seconds=i * step) for i in range(int(span_sec // step) + 1)]
    if (stop_utc - grid[-1]).total_seconds() > 1.0:
        grid.append(stop_utc)

    t_grid = ts.from_datetimes(grid)
    combined = np.asarray(handles[0].visible_at(t_grid), dtype=bool)
    for h in handles[1:]:
        combined = combined & np.asarray(h.visible_at(t_grid), dtype=bool)

    # 提取连续可见区间并二分精化至秒级
    windows: list[AccessWindow] = []
    aer_records: list[AERRecord] = []
    i, n = 0, len(combined)
    acc_num = 0
    while i < n:
        if not combined[i]:
            i += 1
            continue
        j = i
        while j + 1 < n and combined[j + 1]:
            j += 1
        acc_num += 1

        w_start = grid[i]
        if i > 0:
            lo, hi = grid[i - 1], grid[i]
            while (hi - lo).total_seconds() > config.refine_step_sec:
                mid = lo + (hi - lo) / 2
                t_mid = ts.from_datetime(mid)
                if all(bool(h.visible_at(t_mid)) for h in handles):
                    hi = mid
                else:
                    lo = mid
            w_start = hi

        w_stop = grid[j]
        if j < n - 1:
            lo, hi = grid[j], grid[j + 1]
            while (hi - lo).total_seconds() > config.refine_step_sec:
                mid = lo + (hi - lo) / 2
                t_mid = ts.from_datetime(mid)
                if all(bool(h.visible_at(t_mid)) for h in handles):
                    lo = mid
                else:
                    hi = mid
            w_stop = lo

        # 采样窗口内 AER 并定位最高仰角
        dur = max((w_stop - w_start).total_seconds(), 0.0)
        s_step = float(max(config.sample_step_sec, 1.0))
        s_times = [w_start + timedelta(seconds=k * s_step) for k in range(int(dur // s_step) + 1)]
        if (w_stop - s_times[-1]).total_seconds() > 0.5:
            s_times.append(w_stop)
        t_win = ts.from_datetimes(s_times)

        peak_el = -90.0
        peak_time_str = _fmt_bj(w_start)
        for h in handles:
            alt, az, dist = h.altaz(t_win)
            for idx, dt in enumerate(s_times):
                el_val = float(alt.degrees[idx])
                if el_val > peak_el:
                    peak_el = el_val
                    peak_time_str = _fmt_bj(dt)
                aer_records.append(
                    AERRecord(
                        access_number=acc_num,
                        station_name=h.station.name,
                        time=_fmt_bj(dt),
                        azimuth_deg=float(az.degrees[idx]),
                        elevation_deg=el_val,
                        range_km=float(dist.km[idx]),
                    )
                )

        windows.append(
            AccessWindow(
                access_number=acc_num,
                start_time=_fmt_bj(w_start),
                stop_time=_fmt_bj(w_stop),
                duration_sec=dur,
                max_elevation_deg=peak_el,
                max_elevation_time=peak_time_str,
            )
        )
        i = j + 1

    # 星下点轨迹采样
    sub = sat.at(t_grid).subpoint()
    alt0, _, _ = handles[0].altaz(t_grid)
    track = tuple(
        TrackPoint(
            time=_fmt_bj(dt),
            latitude_deg=float(sub.latitude.degrees[idx]),
            longitude_deg=float(sub.longitude.degrees[idx]),
            elevation_deg=float(alt0.degrees[idx]),
            linked=bool(combined[idx]),
        )
```

```python
"""基于 STK 12.6 COM 接口的星地建链与 AER 计算引擎。"""

from __future__ import annotations

from datetime import datetime, timedelta
from zoneinfo import ZoneInfo

from stk_simulation.infrastructure.tle import load_latest_tle
from stk_simulation.simulations.common import OutputResult, _stk_time
from stk_simulation.simulations.link_visibility.config import (
    AERRecord,
    AccessWindow,
    LinkVisibilityConfig,
    LinkVisibilityResult,
)
from stk_simulation.simulations.link_visibility.py_engine import compute_link_visibility_python
from stk_simulation.stk.access import STKAccess
from stk_simulation.stk.constraints import STKConstraints
from stk_simulation.stk.objects import STKObjects
from stk_simulation.stk.session import STKSession

LOCAL_TZ = ZoneInfo("Asia/Shanghai")
TIME_FMT = "%Y-%m-%d %H:%M:%S"


def _fmt_bj(dt_utc: datetime) -> str:
    return dt_utc.astimezone(LOCAL_TZ).strftime(TIME_FMT)


def compute_link_visibility_stk(root, config: LinkVisibilityConfig) -> LinkVisibilityResult:
    """通过 STK 12 计算星地 Access 窗口与 AER 数据。"""
    tle = load_latest_tle(
        config.satellite.tle_url,
        config.paths.data_dir,
        config.satellite.ssc,
    )
    scenario = STKSession.create_scenario(
        root,
        config.scenario.name,
        config.scenario.start_time,
        config.scenario.stop_time,
    )
    sat = STKObjects.create_satellite_from_tle(
        root,
        scenario,
        "SAT_LINK",
        config.satellite.ssc,
        config.satellite.step,
        tle.path,
    )

    active_stations = config.stations[:2] if config.dual_station_sync else config.stations[:1]
    facilities = []
    for idx, st in enumerate(active_stations):
        fac = STKObjects.create_facility(
            scenario, f"Station_{idx + 1}", st.latitude_deg, st.longitude_deg, st.altitude_m / 1000.0
        )
        STKConstraints.set_min_elevation(fac, st.min_elevation_deg)
        facilities.append((st, fac))

    # 获取单站或双站交集窗口
    raw_intervals = STKAccess.compute_access_intervals(facilities[0][1], sat)
    if len(facilities) > 1:
        second_intervals = STKAccess.compute_access_intervals(facilities[1][1], sat)
        intersected: list[tuple[datetime, datetime]] = []
        for s1, e1 in raw_intervals:
            for s2, e2 in second_intervals:
                s_max = max(s1, s2)
                e_min = min(e1, e2)
                if e_min > s_max:
                    intersected.append((s_max, e_min))
        raw_intervals = intersected

    windows: list[AccessWindow] = []
    aer_records: list[AERRecord] = []

    for acc_num, (w_start, w_stop) in enumerate(raw_intervals, start=1):
        start_utcg = _stk_time(w_start.replace(tzinfo=None))
        stop_utcg = _stk_time(w_stop.replace(tzinfo=None))
        peak_el = -90.0
        peak_time = _fmt_bj(w_start)

        for st_cfg, fac_obj in facilities:
            rows = STKAccess.compute_aer_series(
                fac_obj, sat, start_utcg, stop_utcg, config.sample_step_sec
            )
            for row in rows:
                el_val = row["elevation_deg"]
                t_bj = _fmt_bj(row["dt_utc"])
                if el_val > peak_el:
                    peak_el = el_val
                    peak_time = t_bj
                aer_records.append(
                    AERRecord(
                        access_number=acc_num,
                        station_name=st_cfg.name,
                        time=t_bj,
                        azimuth_deg=row["azimuth_deg"],
                        elevation_deg=el_val,
                        range_km=row["range_km"],
                    )
                )

        windows.append(
            AccessWindow(
                access_number=acc_num,
                start_time=_fmt_bj(w_start),
                stop_time=_fmt_bj(w_stop),
                duration_sec=(w_stop - w_start).total_seconds(),
                max_elevation_deg=max(peak_el, 0.0),
                max_elevation_time=peak_time,
            )
        )

    # 复用轻量星下点轨迹用于绘图展示
    py_ref = compute_link_visibility_python(config)
    image_name = "link_visibility_stk.png"
    return LinkVisibilityResult(
        engine_used="stk",
        satellite_name=config.satellite.name,
        station_names=tuple(st.name for st in active_stations),
        start_time=py_ref.start_time,
        stop_time=py_ref.stop_time,
        windows=tuple(windows),
        aer_records=tuple(aer_records),
        ground_track=py_ref.ground_track,
        output=OutputResult(config.paths.result_dir, image_name, config.paths.result_dir / image_name),
        cfg=config,
    )
```

```python
"""星地建链预报统一调度服务（支持 STK / Python / 双引擎交叉精度对比）。"""

from __future__ import annotations

from datetime import datetime
from typing import Callable

from stk_simulation.simulations.link_visibility.config import (
    EngineDiffItem,
    LinkVisibilityConfig,
    LinkVisibilityResult,
)
from stk_simulation.simulations.link_visibility.py_engine import (
    TIME_FMT,
    compute_link_visibility_python,
)
from stk_simulation.simulations.link_visibility.stk_engine import compute_link_visibility_stk


def run_link_visibility(
    root,
    config: LinkVisibilityConfig,
    run_mode: str = "normal",
    progress: Callable[[str], None] | None = None,
) -> LinkVisibilityResult:
    """运行星地建链仿真；run_mode='compare' 时同时跑双引擎并生成秒级精度对比表。"""
    if run_mode == "compare":
        if progress:
            progress("正在运行纯 Python (SGP4) 引擎计算建链窗口...")
        py_res = compute_link_visibility_python(config)
        if progress:
            progress("正在运行 STK 12.6 官方引擎计算 Access 窗口...")
        stk_res = compute_link_visibility_stk(root, config)

        diffs: list[EngineDiffItem] = []
        pair_count = min(len(stk_res.windows), len(py_res.windows))
        for idx in range(pair_count):
            sw = stk_res.windows[idx]
            pw = py_res.windows[idx]
            s_start = datetime.strptime(sw.start_time, TIME_FMT)
            p_start = datetime.strptime(pw.start_time, TIME_FMT)
            s_stop = datetime.strptime(sw.stop_time, TIME_FMT)
            p_stop = datetime.strptime(pw.stop_time, TIME_FMT)
            diffs.append(
                EngineDiffItem(
                    access_number=idx + 1,
                    stk_start=sw.start_time,
                    py_start=pw.start_time,
                    start_diff_sec=(p_start - s_start).total_seconds(),
                    stk_stop=sw.stop_time,
                    py_stop=pw.stop_time,
                    stop_diff_sec=(p_stop - s_stop).total_seconds(),
                    max_el_diff_deg=pw.max_elevation_deg - sw.max_elevation_deg,
                )
            )
        py_res.engine_used = "compare"
        py_res.diff_items = tuple(diffs)
        return py_res

    if getattr(config.stk, "engine", "python") == "stk" and root is not None:
        if progress:
            progress("[STK 引擎] 正在计算星地建链 Access 与 AER...")
        return compute_link_visibility_stk(root, config)

    if progress:
        progress("[Python 引擎] 正在计算 SGP4 星地建链与二分法精化窗口...")
    return compute_link_visibility_python(config)
```

#### 4.1 修改 `src/stk_simulation/app/simulation_controller.py`
升级工作线程控制器，使 `engine == "python"` 时**完全不触碰 STK COM 会话**，只有在 `engine == "stk"` 或 `run_mode == "compare"` 时才按需连接 STK：

```python
"""支持「全球覆盖」与「星地建链」双业务、「STK」与「纯 Python」双引擎的异步工作线程控制器。"""

from __future__ import annotations

import traceback
from typing import TYPE_CHECKING

from PySide6.QtCore import QObject, QThread, Signal, Slot

if TYPE_CHECKING:
    from stk_simulation.stk.session import STKSession


class SimulationWorker(QObject):
    completed = Signal(object, str)  # (result, task_type)
    failed = Signal(str)
    status_changed = Signal(str)

    def __init__(self) -> None:
        super().__init__()
        self._session: STKSession | None = None

    def _ensure_stk_root(self, stk_mode: str):
        if self._session is None:
            from stk_simulation.stk.session import STKSession
            self._session = STKSession()
        return self._session.connect(stk_mode)

    @Slot(str, object, str)
    def run_task(self, task_type: str, config: object, run_mode: str) -> None:
        try:
            engine = getattr(getattr(config, "stk", None), "engine", "python")
            needs_stk = (engine == "stk") or (run_mode == "compare")
            root = self._ensure_stk_root(config.stk.mode) if needs_stk else None

            if task_type == "link_visibility":
                from stk_simulation.simulations.link_visibility.service import (
                    run_link_visibility,
                )
                result = run_link_visibility(
                    root, config, run_mode=run_mode, progress=self.status_changed.emit
                )
            else:
                from stk_simulation.simulations.global_coverage.service import (
                    compute_global_coverage,
                    run_document_scenarios,
                    run_station_gain,
                )
                if run_mode == "document":
                    result = run_document_scenarios(root, config, self.status_changed.emit)
                elif run_mode == "gain":
                    result = run_station_gain(root, config, self.status_changed.emit)
                else:
                    result = compute_global_coverage(
                        root, config, progress=self.status_changed.emit
                    )
        except Exception as error:
            traceback.print_exc()
            self.failed.emit(str(error))
            return
        self.completed.emit(result, task_type)

    @Slot()
    def release(self) -> None:
        if self._session is not None:
            self._session.release()
            self._session = None


class SimulationController(QObject):
    run_requested = Signal(str, object, str)
    release_requested = Signal()
    busy_changed = Signal(bool)
    completed = Signal(object, str)
    failed = Signal(str)
    status_changed = Signal(str)

    def __init__(self, parent: QObject | None = None) -> None:
        super().__init__(parent)
        self._busy = False
        self._thread = QThread(self)
        self._worker = SimulationWorker()
        self._worker.moveToThread(self._thread)
        self.run_requested.connect(self._worker.run_task)
        self.release_requested.connect(self._worker.release)
        self._worker.completed.connect(self._on_completed)
        self._worker.failed.connect(self._on_failed)
        self._worker.status_changed.connect(self.status_changed.emit)
        self._thread.finished.connect(self._worker.deleteLater)
        self._thread.start()

    @property
    def is_busy(self) -> bool:
        return self._busy

    def start(self, config, run_mode: str = "normal", task_type: str = "global_coverage") -> None:
        if self._busy:
            raise RuntimeError("已有仿真任务正在运行，请稍候。")
        self._set_busy(True)
        self.run_requested.emit(task_type, config, run_mode)

    def shutdown(self) -> None:
        self.release_requested.emit()
        self._thread.quit()
        self._thread.wait()

    @Slot(object, str)
    def _on_completed(self, result, task_type: str) -> None:
        self._set_busy(False)
        self.completed.emit(result, task_type)

    @Slot(str)
    def _on_failed(self, message: str) -> None:
        self._set_busy(False)
        self.failed.emit(message)

    def _set_busy(self, busy: bool) -> None:
        self._busy = busy
        self.busy_changed.emit(busy)
```

### Step 6：更新 PyInstaller 打包配置 (`stk_simulation.spec`)

修改 `STKSimulation/GlobalCommunicationSimulation/packaging/stk_simulation.spec`，确保：
1. 即使打包机器未安装 `agi.stk12`，打包过程也不会中断（自动捕获异常降级）；
2. 自动把 `skyfield` 与 `sgp4` 的星历与时间尺度内置数据打包进 EXE；
3. 生成的 EXE 命名为 `SkyOpsSim.exe`：

```python
from pathlib import Path
from PyInstaller.utils.hooks import collect_all, collect_data_files

project_root = Path(SPEC).resolve().parents[1]
source_root = project_root / "src"
app_icon = source_root / "stk_simulation" / "assets" / "app_icon.ico"

datas = []
binaries = []
hiddenimports = ["scipy.spatial.transform._rotation_groups", "skyfield", "sgp4"]

try:
    stk_datas, stk_binaries, stk_hiddenimports = collect_all("agi.stk12")
    datas += list(stk_datas)
    binaries += list(stk_binaries)
    hiddenimports += list(stk_hiddenimports)
except Exception:
    print("[打包提示] 当前环境未检测到 agi.stk12，将仅打包纯 Python 引擎支持。")

for pkg in ("mpl_toolkits.basemap_data", "skyfield", "sgp4"):
    try:
        datas += collect_data_files(pkg)
    except Exception:
        pass

datas.append((str(app_icon), "stk_simulation/assets"))

a = Analysis(
    [str(source_root / "stk_simulation" / "__main__.py")],
    pathex=[str(source_root)],
    binaries=binaries,
    datas=datas,
    hiddenimports=hiddenimports,
    hookspath=[],
    hooksconfig={},
    runtime_hooks=[],
    excludes=["tkinter"],
    noarchive=False,
)
pyz = PYZ(a.pure)

exe = EXE(
    pyz,
    a.scripts,
    a.binaries,
    a.datas,
    [],
    name="SkyOpsSim",
    debug=False,
    bootloader_ignore_signals=False,
    strip=False,
    upx=False,
    console=False,
    disable_windowed_traceback=False,
    icon=str(app_icon),
)
```
